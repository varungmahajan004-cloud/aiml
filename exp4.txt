import pandas as pd
from sklearn.linear_model import LogisticRegression
from sklearn.feature_selection import RFE
from sklearn.preprocessing import StandardScaler

df=pd.read_excel('dataset.xlsx')

# 2. Prepare Data (Exp 03 established this, now we refine it)
# Selecting numerical sensor data as potential features
features = ['Air temperature [K]', 'Process temperature [K]', 
'Rotational speed [rpm]', 'Torque [Nm]', 'Tool wear [min]']
X = df[features]
y = df['Machine failure']

# Standardize features for better performance with Logistic Regression
scaler = StandardScaler()
X_scaled = scaler.fit_transform(X)

# 3. Initialize the Model and RFE (Exp 04 Task: Use RFE with Logistic 
Regression)
# We want to select the top 3 most relevant features
model = LogisticRegression()
rfe = RFE(estimator=model, n_features_to_select=3)
fit = rfe.fit(X_scaled, y)

# 4. Results Analysis
print("Num Features Selected: %d" % fit.n_features_)
print("Selected Features: %s" % [features[i] for i in
range(len(features)) if fit.support_[i]])
print("Feature Ranking: %s" % fit.ranking_)

Num Features Selected: 3
Selected Features: ['Air temperature [K]', 'Rotational speed [rpm]', 
'Torque [Nm]']
Feature Ranking: [1 2 1 1 3]

# Visualization of feature importance
feature_ranking = pd.DataFrame({'Feature': features, 'Ranking': 
fit.ranking_}).sort_values(by='Ranking')
print("\n--- Feature Priority (1 is most relevant) ---")
print(feature_ranking)

 Feature Ranking
